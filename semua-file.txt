<script>
  import { currentRoute, currentPath } from "./router";
  import apiHelper from "./helpers/apiHelper";
  import AuthLayout from "./features/auth/layouts/AuthLayout.svelte";
  import LinkTreeLayout from "./features/linktrees/layouts/LinkTreeLayout.svelte";
  import LoginPage from "./features/auth/pages/LoginPage.svelte";
  import RegisterPage from "./features/auth/pages/RegisterPage.svelte";
  import HomePage from "./features/linktrees/pages/HomePage.svelte";
  import DetailPage from "./features/linktrees/pages/DetailPage.svelte";
  import UsersPage from "./features/users/pages/UsersPage.svelte";
  import ProfilePage from "./features/users/pages/ProfilePage.svelte";
  import NotFoundPage from "./features/common/pages/NotFoundPage.svelte";

  function checkToken() {
    try {
      return !!apiHelper.getAccessToken();
    } catch {
      return false;
    }
  }

  $: tokenExists = checkToken();
  $: activePath = $currentPath || (typeof window !== "undefined" ? window.location.pathname : "/");
  $: routeName = $currentRoute?.name || "home";
  $: routeLayout = $currentRoute?.layout || "linkTree";
</script>

{#if activePath === "/auth/register" || routeName === "register"}
  <AuthLayout>
    <RegisterPage />
  </AuthLayout>
{:else if activePath === "/auth/login" || routeName === "login" || (!tokenExists && routeLayout === "auth")}
  <AuthLayout>
    <LoginPage />
  </AuthLayout>
{:else if activePath === "/" || routeName === "home"}
  <LinkTreeLayout>
    <HomePage />
  </LinkTreeLayout>
{:else if activePath.startsWith("/link-trees") || activePath.startsWith("/todos") || routeName === "linkTree-detail"}
  <LinkTreeLayout>
    <DetailPage params={$currentRoute?.params || {}} />
  </LinkTreeLayout>
{:else if activePath === "/users" || routeName === "users"}
  <LinkTreeLayout>
    <UsersPage />
  </LinkTreeLayout>
{:else if activePath === "/profile" || routeName === "profile"}
  <LinkTreeLayout>
    <ProfilePage />
  </LinkTreeLayout>
{:else}
  <NotFoundPage />
{/if}

  import { Plus, Trash2, Search, ExternalLink, Globe, EyeOff, Lock, User } from "lucide-svelte";
  import { authStore } from "../../auth/states/authStore";
  import { usersStore } from "../../users/states/usersStore";
  import { navigate } from "../../../router";
  import {
    linkTreesStore,
    asyncSetIsLinkTreesGet,
    asyncSetIsLinkTreeDeleteAll
  } from "../states/linkTreesStore";
  import { getImageUrl } from "../../../helpers/toolsHelper";
  import AddModal from "../modals/AddModal.svelte";

  let showAddModal = false;
  let searchQuery = "";
  let onlyMyTrees = false;

  onMount(async () => {
    await Promise.all([asyncSetIsLinkTreesGet(), usersStore.asyncSetUsers()]);
  });

  // Peta id pengguna -> nama, dipakai bila data link tree hanya membawa user_id
  $: userNames = Object.fromEntries(
    (Array.isArray($usersStore?.users) ? $usersStore.users : []).map((u) => [u.id, u.name])
  );

  function ownerName(item, names) {
    return (
      item.user?.name ||
      item.author_name ||
      item.username ||
      names[item.user_id] ||
      (item.user_id ? `User #${item.user_id}` : "Pengguna")
    );
  }

  function openTree(e, id) {
    e.preventDefault();
    navigate(`/link-trees/${id}`);
  }

  $: list = Array.isArray($linkTreesStore)
    ? $linkTreesStore
    : Array.isArray($linkTreesStore?.linkTrees)
      ? $linkTreesStore.linkTrees
      : [];

  $: publicCount = list.filter((item) => item.type === "public").length;
  $: unlistedCount = list.filter((item) => item.type === "unlisted").length;
  $: privateCount = list.filter((item) => item.type === "private").length;

  $: filteredLinkTrees = list.filter((item) => {
    const q = (searchQuery || "").toLowerCase();
    const matchSearch = (item.title || "").toLowerCase().includes(q) ||
                        (item.description || "").toLowerCase().includes(q) ||
                        (item.user?.name || userNames[item.user_id] || "").toLowerCase().includes(q);
    const matchOwner = onlyMyTrees ? item.user_id === $authStore?.profile?.id : true;
    return matchSearch && matchOwner;
  });

  async function handleDeleteAll() {
    if (confirm("Apakah Anda yakin ingin menghapus semua Link Tree?")) {
      if (linkTreesStore.asyncSetIsLinkTreeDeleteAll) {
        await linkTreesStore.asyncSetIsLinkTreeDeleteAll();
      } else {
        await asyncSetIsLinkTreeDeleteAll();
      }
    }
  }
</script>

<script>
  import { onMount } from "svelte";
  import { usersStore } from "../states/usersStore";
  import { Search, Mail, Users, Calendar } from "lucide-svelte";
  import { formatDate } from "../../../helpers/toolsHelper";

  let searchQuery = "";
  let isFetching = true;

  onMount(async () => {
    try {
      await usersStore.asyncSetUsers();
    } finally {
      isFetching = false;
    }
  });

  $: list = Array.isArray($usersStore?.users) ? $usersStore.users : [];

  $: filteredUsers = list.filter((u) => {
    if (!u) return false;
    const q = (searchQuery || "").trim().toLowerCase();
    if (!q) return true;
    const nameMatch = (u.name || "").toLowerCase().includes(q);
    const emailMatch = (u.email || "").toLowerCase().includes(q);
    return nameMatch || emailMatch;
  });

  $: isUsersLoading = isFetching && list.length === 0;
</script>

const BASE_URL =
  typeof DELCOM_BASEURL !== "undefined"
    ? DELCOM_BASEURL
    : "https://open-api.delcom.org/api/v1";

export function getAccessToken() {
  if (typeof window === "undefined" || !window.localStorage) return null;
  return localStorage.getItem("accessToken");
}

export function putAccessToken(token) {
  if (typeof window === "undefined" || !window.localStorage) return;
  if (!token) {
    localStorage.removeItem("accessToken");
  } else {
    localStorage.setItem("accessToken", token);
  }
}

export async function fetchData(endpoint, options = {}) {
  const cleanBase = BASE_URL.replace(/\/+$/, "");
  let cleanEndpoint = endpoint || "";
  if (cleanEndpoint.startsWith("/")) {
    cleanEndpoint = cleanEndpoint.slice(1);
  }
  cleanEndpoint = cleanEndpoint.replace(/\/+$/, "");

  const url = cleanEndpoint.startsWith("http")
    ? cleanEndpoint
    : `${cleanBase}/${cleanEndpoint}`;

  const headers = { ...options.headers };
  const token = getAccessToken();

  if (token) {
    headers["Authorization"] = `Bearer ${token}`;
  }

  if (!(options.body instanceof FormData) && !headers["Content-Type"] && options.method && options.method !== "GET") {
    headers["Content-Type"] = "application/json";
  }

  const config = {
    ...options,
    headers,
  };

  return await fetch(url, config);
}

async function request(endpoint, options = {}) {
  const response = await fetchData(endpoint, options);
  const data = await response.json().catch(() => ({}));

  if (!response.ok) {
    const error = new Error(data.message || `Request failed with status ${response.status}`);
    error.status = response.status;
    error.data = data;
    throw error;
  }

  return data;
}

export const apiHelper = {
  fetchData,
  get: (endpoint, options) => request(endpoint, { ...options, method: "GET" }),
  post: (endpoint, body, options) =>
    request(endpoint, {
      ...options,
      method: "POST",
      body: body instanceof FormData ? body : JSON.stringify(body),
    }),
  put: (endpoint, body, options) =>
    request(endpoint, {
      ...options,
      method: "PUT",
      body: body instanceof FormData ? body : JSON.stringify(body),
    }),
  delete: (endpoint, options) =>
    request(endpoint, { ...options, method: "DELETE" }),
  getAccessToken,
  putAccessToken,
};

export default apiHelper;

export function formatDate(date) {
  if (!date) return "-";
  return new Date(date).toLocaleString("id-ID", {
    day: "2-digit",
    month: "long",
    year: "numeric",
    hour: "2-digit",
    minute: "2-digit",
  });
}

export function getImageUrl(path) {
  if (!path) return null;
  if (path.startsWith("http://") || path.startsWith("https://") || path.startsWith("data:")) {
    return path;
  }
  const baseUrl = (typeof DELCOM_BASEURL !== "undefined" ? DELCOM_BASEURL : "https://open-api.delcom.org/api/v1")
    .replace(/\/api\/v1\/?$/, "");
  const cleanPath = path.startsWith("/") ? path : `/${path}`;
  return `${baseUrl}${cleanPath}`;
}
