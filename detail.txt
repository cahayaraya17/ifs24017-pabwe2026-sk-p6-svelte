<script>
  import { onMount } from "svelte";
  import {
    Edit2,
    Trash2,
    Image,
    Plus,
    ExternalLink,
    ArrowLeft
  } from "lucide-svelte";
  import { navigate } from "../../../router";
  import {
    linkTreesStore,
    linkTreeDetailStore,
    asyncSetIsLinkTreeGet,
    asyncSetIsLinkTreeDelete,
    asyncSetIsItemDelete
  } from "../states/linkTreesStore";
  import MarkdownViewer from "../components/MarkdownViewer.svelte";
  import ChangeModal from "../modals/ChangeModal.svelte";
  import ChangeCoverModal from "../modals/ChangeCoverModal.svelte";
  import ItemModal from "../modals/ItemModal.svelte";

  export let params = {};

  let showChangeModal = false;
  let showChangeCoverModal = false;
  let showItemModal = false;
  let selectedItem = null;
  let isLoading = true;

  $: linkTreeId = params.linkTreeId || params.id;
  $: linkTree = $linkTreeDetailStore || $linkTreesStore.linkTree;

  onMount(async () => {
    try {
      if (linkTreeId) {
        await asyncSetIsLinkTreeGet(linkTreeId);
      }
    } finally {
      isLoading = false;
    }
  });

  async function handleDeleteTree() {
    if (confirm("Apakah Anda yakin ingin menghapus Link Tree ini?")) {
      const success = await asyncSetIsLinkTreeDelete(linkTreeId);
      if (success) {
        navigate("/");
      }
    }
  }

  function handleAddItem() {
    selectedItem = null;
    showItemModal = true;
  }

  function handleEditItem(item) {
    selectedItem = item;
    showItemModal = true;
  }

  async function handleDeleteItem(itemId) {
    if (confirm("Apakah Anda yakin ingin menghapus tautan ini?")) {
      await asyncSetIsItemDelete(linkTreeId, itemId);
    }
  }
</script>

<div class="space-y-6">
  <div>
    <a
      href="#/"
      on:click|preventDefault={() => navigate("/") }
      class="inline-flex items-center text-sm font-medium text-gray-500 hover:text-gray-700 dark:text-gray-400 dark:hover:text-gray-200"
    >
      <ArrowLeft class="mr-1.5 h-4 w-4" />
      Kembali ke Semua Link Tree
    </a>
  </div>

  {#if linkTree}
    <div class="overflow-hidden rounded-xl border border-gray-200 bg-white shadow-sm dark:border-gray-700 dark:bg-gray-800">
      {#if linkTree.cover}
        <div class="h-48 w-full overflow-hidden bg-gray-100 dark:bg-gray-900">
          <img src={linkTree.cover} alt={linkTree.title} class="h-full w-full object-cover" />
        </div>
      {/if}

      <div class="p-6">
        <div class="flex flex-col gap-4 sm:flex-row sm:items-start sm:justify-between">
          <div>
            <div class="flex items-center gap-2">
              <h1 class="text-2xl font-bold text-gray-900 dark:text-white">{linkTree.title}</h1>
              <span
                class="rounded-full px-2.5 py-0.5 text-xs font-medium uppercase {
                  linkTree.type === 'public'
                    ? 'bg-blue-100 text-blue-700 dark:bg-blue-900/40 dark:text-blue-300'
                    : linkTree.type === 'unlisted'
                      ? 'bg-amber-100 text-amber-700 dark:bg-amber-900/40 dark:text-amber-300'
                      : 'bg-red-100 text-red-700 dark:bg-red-900/40 dark:text-red-300'
                }"
              >
                {linkTree.type || "public"}
              </span>
            </div>
            {#if linkTree.description}
              <div class="mt-3 text-gray-600 dark:text-gray-300">
                <MarkdownViewer source={linkTree.description} />
              </div>
            {/if}
          </div>

          <div class="flex flex-wrap items-center gap-2">
            <button
              type="button"
              class="inline-flex items-center rounded-lg border border-gray-300 bg-white px-3 py-2 text-sm font-medium text-gray-700 hover:bg-gray-50 dark:border-gray-600 dark:bg-gray-700 dark:text-gray-200"
              on:click={() => (showChangeCoverModal = true)}
            >
              <Image class="mr-1.5 h-4 w-4" />
              Ganti Sampul
            </button>
            <button
              type="button"
              class="inline-flex items-center rounded-lg border border-gray-300 bg-white px-3 py-2 text-sm font-medium text-gray-700 hover:bg-gray-50 dark:border-gray-600 dark:bg-gray-700 dark:text-gray-200"
              on:click={() => (showChangeModal = true)}
            >
              <Edit2 class="mr-1.5 h-4 w-4" />
              Ubah
            </button>
            <button
              type="button"
              class="inline-flex items-center rounded-lg border border-red-300 bg-white px-3 py-2 text-sm font-medium text-red-600 hover:bg-red-50 dark:border-red-900 dark:bg-gray-800 dark:hover:bg-red-950/30"
              on:click={handleDeleteTree}
            >
              <Trash2 class="mr-1.5 h-4 w-4" />
              Hapus
            </button>
          </div>
        </div>
      </div>
    </div>

    <div class="space-y-4">
      <div class="flex items-center justify-between">
        <h2 class="text-lg font-bold text-gray-900 dark:text-white">Daftar Tautan</h2>
        <button
          type="button"
          class="inline-flex items-center rounded-lg bg-blue-600 px-4 py-2 text-sm font-medium text-white hover:bg-blue-700"
          on:click={handleAddItem}
        >
          <Plus class="mr-1.5 h-4 w-4" />
          Tambah Tautan
        </button>
      </div>

      {#if !linkTree.items || linkTree.items.length === 0}
        <div class="rounded-xl border border-dashed border-gray-300 p-8 text-center dark:border-gray-700">
          <p class="text-sm text-gray-500 dark:text-gray-400">Belum ada item tautan yang ditambahkan.</p>
        </div>
      {:else}
        <div class="space-y-3">
          {#each linkTree.items as item (item.id)}
            <div class="flex items-center justify-between rounded-xl border border-gray-200 bg-white p-4 shadow-sm dark:border-gray-700 dark:bg-gray-800">
              <div class="min-w-0 flex-1 pr-4">
                <h4 class="truncate font-semibold text-gray-900 dark:text-white">{item.title}</h4>
                <a
                  href={item.url}
                  target="_blank"
                  rel="noreferrer"
                  class="inline-flex items-center truncate text-sm text-blue-600 hover:underline dark:text-blue-400"
                >
                  {item.url}
                  <ExternalLink class="ml-1 h-3.5 w-3.5 shrink-0" />
                </a>
              </div>
              <div class="flex shrink-0 items-center gap-1">
                <button
                  type="button"
                  class="rounded-lg p-2 text-gray-500 hover:bg-gray-100 hover:text-gray-700 dark:text-gray-400 dark:hover:bg-gray-700 dark:hover:text-gray-200"
                  on:click={() => handleEditItem(item)}
                  title="Ubah"
                >
                  <Edit2 class="h-4 w-4" />
                </button>
                <button
                  type="button"
                  class="rounded-lg p-2 text-red-500 hover:bg-red-50 hover:text-red-700 dark:hover:bg-red-950/30"
                  on:click={() => handleDeleteItem(item.id)}
                  title="Hapus"
                >
                  <Trash2 class="h-4 w-4" />
                </button>
              </div>
            </div>
          {/each}
        </div>
      {/if}
    </div>
  {:else if isLoading}
    <div class="rounded-xl border border-gray-200 bg-white p-12 text-center dark:border-gray-700 dark:bg-gray-800">
      <p class="text-gray-500 dark:text-gray-400">Memuat data Link Tree...</p>
    </div>
  {:else}
    <div class="rounded-xl border border-gray-200 bg-white p-12 text-center dark:border-gray-700 dark:bg-gray-800">
      <p class="text-gray-500 dark:text-gray-400">Link Tree tidak ditemukan atau tidak dapat diakses.</p>
    </div>
  {/if}
</div>

<ChangeModal show={showChangeModal} linkTree={linkTree} on:close={() => (showChangeModal = false)} />
<ChangeCoverModal show={showChangeCoverModal} linkTreeId={linkTreeId} on:close={() => (showChangeCoverModal = false)} />
<ItemModal show={showItemModal} linkTreeId={linkTreeId} item={selectedItem} on:close={() => (showItemModal = false)} />

import { describe, it, expect, vi, beforeEach, afterEach } from "vitest";
import { render, fireEvent, waitFor, cleanup } from "@testing-library/svelte";
import DetailPage from "./DetailPage.svelte";
import * as store from "../states/linkTreesStore";
import * as router from "../../../router";

vi.mock("../modals/ChangeModal.svelte", () => import("./__stubs__/ModalStub.svelte"));
vi.mock("../modals/ChangeCoverModal.svelte", () => import("./__stubs__/ModalStub.svelte"));
vi.mock("../modals/ItemModal.svelte", () => import("./__stubs__/ModalStub.svelte"));

const makeTree = (o = {}) => ({
  id: 1,
  title: "My Tree",
  type: "public",
  items: [{ id: 100, title: "My Portfolio", url: "https://portfolio.com" }],
  ...o,
});

let navSpy, getSpy, delTreeSpy, delItemSpy;

function setup(tree, props = { params: { linkTreeId: 1 } }) {
  store.linkTreeDetailStore.set(tree);
  return render(DetailPage, { props });
}

describe("DetailPage - tambahan coverage", () => {
  beforeEach(() => {
    navSpy = vi.spyOn(router, "navigate").mockImplementation(() => {});
    getSpy = vi.spyOn(store, "asyncSetIsLinkTreeGet").mockResolvedValue(true);
    delTreeSpy = vi.spyOn(store, "asyncSetIsLinkTreeDelete").mockResolvedValue(true);
    delItemSpy = vi.spyOn(store, "asyncSetIsItemDelete").mockResolvedValue(true);
    store.linkTreeDetailStore.set(null);
    store.linkTreesStore.set({ isLinkTree: false, linkTree: null });
  });

  afterEach(() => {
    cleanup();
    vi.restoreAllMocks();
  });

  describe("params & onMount", () => {
    it("tanpa props: tidak memanggil get", () => {
      render(DetailPage);
      expect(getSpy).not.toHaveBeenCalled();
    });

    it("memakai params.id", async () => {
      render(DetailPage, { props: { params: { id: 7 } } });
      await waitFor(() => expect(getSpy).toHaveBeenCalledWith(7));
    });

    it("memakai params.linkTreeId", async () => {
      render(DetailPage, { props: { params: { linkTreeId: 3 } } });
      await waitFor(() => expect(getSpy).toHaveBeenCalledWith(3));
    });
  });

  describe("state tampilan", () => {
    it("menampilkan pesan memuat", () => {
      const { getByText } = render(DetailPage);
      expect(getByText("Memuat data Link Tree...")).toBeTruthy();
    });

    it("menampilkan pesan memuat saat linkTreesStore null", () => {
      store.linkTreesStore.set(null);
      const { getByText } = render(DetailPage);
      expect(getByText("Memuat data Link Tree...")).toBeTruthy();
    });

    it("menampilkan pesan tidak ditemukan", () => {
      store.linkTreesStore.set({ isLinkTree: true, linkTree: null });
      const { getByText } = render(DetailPage);
      expect(getByText(/tidak ditemukan/)).toBeTruthy();
    });

    it("fallback ke linkTreesStore.linkTree", () => {
      store.linkTreesStore.set({ isLinkTree: true, linkTree: makeTree({ title: "Dari Store" }) });
      const { getByText } = render(DetailPage);
      expect(getByText("Dari Store")).toBeTruthy();
    });

    it("menampilkan cover", () => {
      const { container } = setup(makeTree({ cover: "cover.png" }));
      const img = container.querySelector("img");
      expect(img).toBeTruthy();
      expect(img.getAttribute("alt")).toBe("My Tree");
    });

    it("tanpa cover tidak ada img", () => {
      const { container } = setup(makeTree());
      expect(container.querySelector("img")).toBeNull();
    });

    it("tipe unlisted", () => {
      const { getByText } = setup(makeTree({ type: "unlisted" }));
      expect(getByText("unlisted")).toBeTruthy();
    });

    it("tipe private", () => {
      const { getByText } = setup(makeTree({ type: "private" }));
      expect(getByText("private")).toBeTruthy();
    });

    it("tipe kosong jatuh ke public", () => {
      const { getByText } = setup(makeTree({ type: undefined }));
      expect(getByText("public")).toBeTruthy();
    });

    it("menampilkan deskripsi", () => {
      const { container } = setup(makeTree({ description: "Bio saya" }));
      expect(container.querySelector(".mt-3")).toBeTruthy();
    });

    it("tanpa deskripsi", () => {
      const { container } = setup(makeTree({ description: "" }));
      expect(container.querySelector(".mt-3")).toBeNull();
    });

    it("items undefined menampilkan pesan kosong", () => {
      const { getByText } = setup(makeTree({ items: undefined }));
      expect(getByText("Belum ada item tautan yang ditambahkan.")).toBeTruthy();
    });

    it("items kosong menampilkan pesan kosong", () => {
      const { getByText } = setup(makeTree({ items: [] }));
      expect(getByText("Belum ada item tautan yang ditambahkan.")).toBeTruthy();
    });
  });

  describe("navigasi", () => {
    it("link kembali memanggil navigate('/')", async () => {
      const { getByText } = setup(makeTree());
      await fireEvent.click(getByText("Kembali ke Semua Link Tree"));
      expect(navSpy).toHaveBeenCalledWith("/");
    });
  });

  describe("hapus link tree", () => {
    it("dikonfirmasi & sukses: navigate ke /", async () => {
      vi.spyOn(window, "confirm").mockReturnValue(true);
      const { getAllByText } = setup(makeTree());
      await fireEvent.click(getAllByText("Hapus")[0]);
      await waitFor(() => expect(navSpy).toHaveBeenCalledWith("/"));
      expect(delTreeSpy).toHaveBeenCalledWith(1);
    });

    it("dikonfirmasi tapi gagal: tidak navigate", async () => {
      vi.spyOn(window, "confirm").mockReturnValue(true);
      delTreeSpy.mockResolvedValue(false);
      const { getAllByText } = setup(makeTree());
      await fireEvent.click(getAllByText("Hapus")[0]);
      await waitFor(() => expect(delTreeSpy).toHaveBeenCalled());
      expect(navSpy).not.toHaveBeenCalled();
    });

    it("dibatalkan: tidak menghapus", async () => {
      vi.spyOn(window, "confirm").mockReturnValue(false);
      const { getAllByText } = setup(makeTree());
      await fireEvent.click(getAllByText("Hapus")[0]);
      expect(delTreeSpy).not.toHaveBeenCalled();
    });
  });

  describe("item", () => {
    it("tambah item membuka modal lalu menutupnya", async () => {
      const { getByText, queryByTestId, getByTestId } = setup(makeTree());
      expect(queryByTestId("stub-close")).toBeNull();
      await fireEvent.click(getByText("Tambah Tautan"));
      await fireEvent.click(getByTestId("stub-close"));
      expect(queryByTestId("stub-close")).toBeNull();
    });

    it("ubah item membuka modal lalu menutupnya", async () => {
      const { getByTitle, queryByTestId, getByTestId } = setup(makeTree());
      await fireEvent.click(getByTitle("Ubah"));
      await fireEvent.click(getByTestId("stub-close"));
      expect(queryByTestId("stub-close")).toBeNull();
    });

    it("hapus item dibatalkan", async () => {
      vi.spyOn(window, "confirm").mockReturnValue(false);
      const { getByTitle } = setup(makeTree());
      await fireEvent.click(getByTitle("Hapus"));
      expect(delItemSpy).not.toHaveBeenCalled();
    });
  });

  describe("modal tree", () => {
    it("modal Ganti Sampul dibuka dan ditutup", async () => {
      const { getAllByText, queryByTestId, getByTestId } = setup(makeTree());
      await fireEvent.click(getAllByText("Ganti Sampul")[0]);
      await fireEvent.click(getByTestId("stub-close"));
      expect(queryByTestId("stub-close")).toBeNull();
    });

    it("modal Ubah dibuka dan ditutup", async () => {
      const { getAllByText, queryByTestId, getByTestId } = setup(makeTree());
      await fireEvent.click(getAllByText("Ubah")[0]);
      await fireEvent.click(getByTestId("stub-close"));
      expect(queryByTestId("stub-close")).toBeNull();
    });
  });
});
