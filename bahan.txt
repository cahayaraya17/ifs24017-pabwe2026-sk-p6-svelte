<script>
  import { createEventDispatcher } from "svelte";
  import { ImagePlus, X, LoaderCircle, Upload } from "lucide-svelte";
  import { linkTreesStore } from "../states/link-treesStore";
  import { showErrorDialog } from "../../../helpers/toolsHelper";

  export let show = false;
  export let linkTree = null;
  export let onClose = undefined;

  const dispatch = createEventDispatcher();

  let loading = false;
  let file = null;
  let previewUrl = null;

  function handleClose() {
    previewUrl = null;
    file = null;
    dispatch("close");
    if (onClose) onClose();
  }

  $: {
    if (show) {
      document.body.style.overflow = "hidden";
    } else {
      document.body.style.overflow = "auto";
    }
  }

  $: if ($linkTreesStore.isLinkTreeChangeCover) {
    loading = false;
    linkTreesStore.setIsLinkTreeChangeCover(false);
    if ($linkTreesStore.isLinkTreeChangedCover) {
      linkTreesStore.setIsLinkTreeChangedCover(false);
      linkTreesStore.asyncSetLinkTree(linkTree.id);
      handleClose();
    }
  }

  function handleFileChange(e) {
    const selectedFile = e.target.files?.[0];
    if (!selectedFile) return;

    if (!selectedFile.type.startsWith("image/")) {
      showErrorDialog("Pilih file gambar (JPG/PNG)!");
      return;
    }

    if (selectedFile.size > 1024 * 1024) {
      showErrorDialog("Ukuran gambar maksimal 1MB!");
      return;
    }

    file = selectedFile;
    previewUrl = URL.createObjectURL(selectedFile);
  }

  function handleSave() {
    if (!file) {
      showErrorDialog("Silakan pilih cover baru!");
      return;
    }

    loading = true;
    linkTreesStore.asyncSetIsLinkTreeChangeCover(linkTree.id, file);
  }
</script>

{#if show && linkTree}
  <div
    data-testid="change-cover-modal"
    class="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-900/50 backdrop-blur-xs animate-in fade-in duration-200"

        <div class="flex items-center gap-2.5">
          <div class="w-8 h-8 rounded-lg bg-sky-100 text-sky-700 flex items-center justify-center">
            <ImagePlus size={18} strokeWidth={2.5} />
          </div>

        </div>
        <button

          on:click={handleClose}
          class="p-1.5 rounded-lg text-slate-400 hover:text-slate-600 hover:bg-slate-100 transition-colors"
        >
          <X size={18} />

      </div>

  import { authStore } from "../../auth/states/authStore";

  let isMobileSidebarOpen = false;

  let hasToken = !!apiHelper.getAccessToken();

  onMount(async () => {
    isMounted = true;
    hasToken = !!apiHelper.getAccessToken();

    if (!hasToken) {
      navigate("/auth/login");
      return;
    }

    if (!profile) {
      await usersStore.asyncSetProfile();
    }
  });

  $: if ($authStore?.isAuthLogout) {
    navigate("/auth/login");
  }

  $: if ($usersStore?.isProfile && !profile && isMounted) {
    apiHelper.putAccessToken("");
    navigate("/auth/login");
  }

  function handleToggleMobileSidebar() {
    isMobileSidebarOpen = !isMobileSidebarOpen;
  }

  function handleCloseMobileSidebar() {
    isMobileSidebarOpen = false;
  }

  async function handleLogout() {
    await authStore.asyncSetIsAuthLogout();
  }
</script>

  // Peta id pengguna -> nama, dipakai bila data link tree hanya membawa user_id
  $: userNames = Object.fromEntries(
    (Array.isArray($usersStore?.users) ? $usersStore.users : []).map((u) => [u.id, u.name])
  );

  function ownerName(item, names) {
    return (
      item.user?.name ||
      item.author_name ||
      item.username ||
      names[item.user_id] ||
      (item.user_id ? `User #${item.user_id}` : "Pengguna")
    );
  }

  function openTree(e, id) {
    e.preventDefault();
    navigate(`/link-trees/${id}`);
  }

  $: list = Array.isArray($linkTreesStore)
    ? $linkTreesStore
    : Array.isArray($linkTreesStore?.linkTrees)
      ? $linkTreesStore.linkTrees
      : [];

  $: publicCount = list.filter((item) => item.type === "public").length;
  $: unlistedCount = list.filter((item) => item.type === "unlisted").length;
  $: privateCount = list.filter((item) => item.type === "private").length;

  $: filteredLinkTrees = list.filter((item) => {
    const q = (searchQuery || "").toLowerCase();
    const matchSearch = (item.title || "").toLowerCase().includes(q) ||
                        (item.description || "").toLowerCase().includes(q) ||
                        (item.user?.name || userNames[item.user_id] || "").toLowerCase().includes(q);
    const matchOwner = onlyMyTrees ? item.user_id === $authStore?.profile?.id : true;
    return matchSearch && matchOwner;
  });

  async function handleDeleteAll() {
    if (confirm("Apakah Anda yakin ingin menghapus semua Link Tree?")) {
      if (linkTreesStore.asyncSetIsLinkTreeDeleteAll) {
        await linkTreesStore.asyncSetIsLinkTreeDeleteAll();
      } else {
        await asyncSetIsLinkTreeDeleteAll();
      }
    }

import apiHelper from "../../../helpers/apiHelper";

const linkTreeApi = {
  async postLinkTree(title, type = "public", description = "") {
    try {
      const response = await apiHelper.post("/link-trees", {
        title,
        type,
        description,
      });
      return response.data !== undefined ? response.data : response;
    } catch (error) {
      if (!error || !error.message) {
        throw new Error("Gagal menambahkan linkTree");
      }
